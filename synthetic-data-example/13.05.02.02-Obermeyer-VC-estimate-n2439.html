Objective

I want to propose a test for variance differences in the calibration model. i.e. I propose to fit the following GAM: \(Y_i = \mu(r_i) + \Delta(r_i)I(s_i = s) + \varepsilon_{i}\) where \(\varepsilon_i | s_i = s \sim N(0,\sigma_s^2)\)

Then want to test \(\sigma_1^2 = \sigma_2^2\)

# ---- Select pilot bandwidth for local cubic polynomial to minimize IRSC ---- #
rsc.start <- Sys.time()
# defining h grid using Fan and Gijbels approach:
U.max = max(U); U.min = min(U); n = length(u)
h.min <- (U.max - U.min)/n # this is too small for many u to invert all matrices so:
h.min <- max(h.min, 0.025*(U.max - U.min))
h.max <- (U.max - U.min)/2
h.grid <- seq(h.min, h.max, 0.01)
adj <- 0.7776

rsc.u0 <- NULL
irsc.h <- NULL
for(h in h.grid){
  rsc.h <- NULL
  for(x in U){
    r <- rsc(Y,s,u, u0=x, q = 1, a = 2, h=h)
    rsc.h <- rbind(rsc.h, c(x, h, r)) 
    colnames(rsc.h) <- c("u0","h","RSC")
  }
  irsc <- mean(rsc.h[,"RSC"])
  irsc.h <- rbind(irsc.h, c(h, irsc)) 
  colnames(irsc.h) <- c("h","IRSC")
  rsc.u0 <- rbind(rsc.u0, rsc.h)
}

h.star <- irsc.h %>% as.data.frame() %>% slice_min(IRSC, n = 1 ) %>% pull(h)
h.rsc <- h.star*adj
